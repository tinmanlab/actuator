option(QDD_SANITIZE "Enable host address/undefined behavior sanitizers" OFF)
option(QDD_HOST "Build host simulator and tests" ON)
add_library(qdd_core src/control.cpp src/c_api.cpp)
target_include_directories(qdd_core PUBLIC include)
if(MSVC)
 target_compile_options(qdd_core PRIVATE /W4 /fp:precise)
else()
 add_compile_options(-Wall -Wextra -Wpedantic -Werror)
 target_compile_options(qdd_core PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions)
 if(NOT QDD_SANITIZE)
  target_compile_options(qdd_core PRIVATE -fno-rtti)
 endif()
endif()
if(QDD_SANITIZE AND NOT MSVC)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
 target_compile_options(qdd_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
if(QDD_HOST)
 # Bind runtime results to the source used for this build. Reconfigure on content changes.
 file(GLOB_RECURSE QDD_HASH_FILES CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}"
  "src/*.cpp" "include/*.hpp" "include/*.h" "apps/*.cpp" "cmake/*.cmake")
 list(APPEND QDD_HASH_FILES "CMakeLists.txt")
 list(SORT QDD_HASH_FILES)
 set(QDD_SOURCE_MANIFEST "")
 foreach(path IN LISTS QDD_HASH_FILES)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${path}" content_hash)
  string(APPEND QDD_SOURCE_MANIFEST "${path}:${content_hash}\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${path}")
 endforeach()
 string(SHA256 QDD_SOURCE_DIGEST "${QDD_SOURCE_MANIFEST}")
 add_library(qdd_host src/plant.cpp src/bench.cpp src/config_snapshot.cpp src/protection.cpp)
 target_compile_definitions(qdd_host PRIVATE QDD_SOURCE_SHA256="${QDD_SOURCE_DIGEST}")
 target_link_libraries(qdd_host PUBLIC qdd_core)
 add_executable(qdd_sim apps/qdd_sim.cpp)
 target_link_libraries(qdd_sim PRIVATE qdd_host)
 include(CTest)
 if(BUILD_TESTING)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS tests/test_core.cpp tests/test_v03.cpp)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
   add_test(NAME fast_trip_analysis_fixtures COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_fast_trip_analysis.py)
   add_test(NAME analysis_analytic_fixtures COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_analysis.py)
   add_test(NAME correlation_contract_fixtures COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_correlation.py)
   add_test(NAME native_cli_evidence_workflow COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_workflow.py --exe $<TARGET_FILE:qdd_sim>)
  endif()
  add_executable(qdd_tests tests/test_core.cpp)
  target_link_libraries(qdd_tests PRIVATE qdd_host)
  add_executable(qdd_c_test tests/test_c_api.c)
  target_link_libraries(qdd_c_test PRIVATE qdd_core)
  set_target_properties(qdd_c_test PROPERTIES LINKER_LANGUAGE CXX)
  add_test(NAME c11_caller COMMAND qdd_c_test)
  add_executable(qdd_port_test tests/test_port.cpp)
  target_link_libraries(qdd_port_test PRIVATE qdd_core)
  add_test(NAME stm32_port_host_contract COMMAND qdd_port_test)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS tests/test_v04.cpp)
  add_executable(qdd_v04_tests tests/test_v04.cpp)
  target_link_libraries(qdd_v04_tests PRIVATE qdd_host)
  file(STRINGS tests/test_v04.cpp V04_TEST_LINES REGEX "tests\\[\"")
  foreach(line IN LISTS V04_TEST_LINES)
   string(REGEX REPLACE ".*tests\\[\"([^\"]+)\"\\].*" "\\1" test_name "${line}")
   add_test(NAME ${test_name} COMMAND qdd_v04_tests ${test_name})
  endforeach()
  add_executable(qdd_v03_tests tests/test_v03.cpp)
  target_link_libraries(qdd_v03_tests PRIVATE qdd_host)
  file(STRINGS tests/test_v03.cpp V03_TEST_LINES REGEX "tests\\[\"")
  foreach(line IN LISTS V03_TEST_LINES)
   string(REGEX REPLACE ".*tests\\[\"([^\"]+)\"\\].*" "\\1" test_name "${line}")
   add_test(NAME ${test_name} COMMAND qdd_v03_tests ${test_name})
  endforeach()
  file(STRINGS tests/test_core.cpp QDD_TEST_LINES REGEX "tests\\[\"")
  foreach(line IN LISTS QDD_TEST_LINES)
   string(REGEX REPLACE ".*tests\\[\"([^\"]+)\"\\].*" "\\1" test_name "${line}")
   add_test(NAME ${test_name} COMMAND qdd_tests ${test_name})
  endforeach()
 endif()
endif()
